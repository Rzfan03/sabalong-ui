### Usage
```tsx
import { Modal } from "sabalong-ui";

<Modal
  open={isOpen}
  onClose={() => setIsOpen(false)}
  title="Confirm"
>
  Are you sure?
</Modal>
```

Closes on backdrop click, the X button, or `Escape`.
